+# Task 1 implementation report
+
+Implemented the meeting persistence and lifecycle layer using the shared Task 1 contracts. Added the seven Prisma models and additive SQLite migration, lifecycle constants/errors/policies, and an injected-Prisma meeting service. Active source identity is protected by the composite `(meetingId, activeHash)` unique key; meeting import identity uses a nullable unique `activeKey`. Metadata creation checks active meetings and active import reservations in a transaction. Mutations check lifecycle state and invoke the audit callback.
+
+The business timezone is `Asia/Shanghai`. Quarter assignment uses the local month in that timezone. Meeting/transcript time intervals use `[from, to)`, with `from` included and `to` excluded. All persisted JSON-like contract fields remain serialized strings.

+function fakePrisma(seed = {}) {
+  const state = {
+    meetings: [...(seed.meetings || [])],
+    meeting_sources: [...(seed.meeting_sources || [])],
+    meeting_chunks: [...(seed.meeting_chunks || [])],
+    meeting_jobs: [...(seed.meeting_jobs || [])],
+    nextMeetingId: 1,
+    nextJobId: 1,
+  };
+  const matches = (row, where = {}) => Object.entries(where).every(([key, value]) => {
+    if (value && typeof value === "object" && "in" in value) return value.in.includes(row[key]);
+    return row[key] === value;
+  });
+  const delegate = (collection) => ({
+    findMany: async ({ where = {} } = {}) => state[collection].filter((row) => matches(row, where)),
+    findFirst: async ({ where = {} } = {}) => state[collection].find((row) => matches(row, where)) || null,
+    findUnique: async ({ where = {} } = {}) => state[collection].find((row) => matches(row, where)) || null,
+    create: async ({ data }) => {
+      const row = { id: collection === "meetings" ? state.nextMeetingId++ : state.nextJobId++, ...data };
+      state[collection].push(row);
+      return row;
+    },
+    update: async ({ where, data }) => {
+      const row = state[collection].find((candidate) => matches(candidate, where));
+      if (!row) throw new Error("record not found");
+      Object.assign(row, data);
+      return row;
+    },
+    updateMany: async ({ where = {}, data }) => {
+      const rows = state[collection].filter((row) => matches(row, where));
+      rows.forEach((row) => Object.assign(row, data));
+      return { count: rows.length };
+    },
+    deleteMany: async ({ where = {} } = {}) => {
+      const rows = state[collection].filter((row) => matches(row, where));
+      state[collection] = state[collection].filter((row) => !rows.includes(row));
+      return { count: rows.length };
+    },
+  });
+  const db = {
+    meetings: delegate("meetings"),
+    meeting_sources: delegate("meeting_sources"),
+    meeting_chunks: delegate("meeting_chunks"),
+    meeting_jobs: delegate("meeting_jobs"),
+    $transaction: async (callback) => callback(db),
+  };
+  return { db, state };
+}

+const { MeetingError } = require("./errors");
+const { LIMITS, REVIEW, RETRIEVAL, quarterFor } = require("./policy");
+
+const activeMeetingWhere = { deletedAt: null };
+const now = () => new Date();
+
+function createMeetingService({ prisma, audit = async () => {} } = {}) {
+  if (!prisma) throw new TypeError("createMeetingService requires an injected Prisma client");
+
+  async function transaction(callback) {
+    return typeof prisma.$transaction === "function" ? prisma.$transaction(callback) : callback(prisma);
+  }
+
+  async function findMeeting(db, id) {
+    const meeting = await db.meetings.findUnique({ where: { id: Number(id) } });
+    if (!meeting || meeting.retrievalStatus === RETRIEVAL.DELETED) {
+      throw new MeetingError("Meeting not found", "MEETING_NOT_FOUND", 404);
+    }
+    return meeting;
+  }
+
+  async function auditMutation(action, meeting, userId, details = {}) {
+    await audit({ action, meetingId: meeting.id, userId: userId ?? null, ...details });
+  }
+
+  return {
+    async list(filters = {}) {
+      return prisma.meetings.findMany({ where: { ...filters, ...activeMeetingWhere } });
+    },
+
+    async get(id) {
+      return findMeeting(prisma, id);
+    },
+
+    async createMetadata(data, userId) {
+      const startTime = new Date(data.startTime);
+      if (Number.isNaN(startTime.getTime())) throw new MeetingError("A valid meeting start time is required", "INVALID_DATE");
+      const quarter = data.quarter || quarterFor(startTime);
+      const activeKey = `${data.connectionId}:${data.externalId}`;
+      try {
+        return await transaction(async (db) => {
+          const duplicate = await db.meetings.findUnique({ where: { activeKey } });
+          if (duplicate) throw new MeetingError("This meeting is already imported", "MEETING_DUPLICATE", 409);
+
+          const existing = await db.meetings.findMany({ where: { quarter, deletedAt: null } });
+          const activeKeys = new Set(existing.map((meeting) => meeting.activeKey).filter(Boolean));
+          const reservations = await db.meeting_jobs.findMany({
+            where: { quarter, kind: "IMPORT", status: { in: ["CREATED", "PROCESSING"] } },
+          });
+          const reservationCount = reservations.filter((job) => !job.activeKey || !activeKeys.has(job.activeKey)).length;
+          if (existing.length + reservationCount >= LIMITS.meetingsPerQuarter) {
+            throw new MeetingError("The quarterly meeting limit has been reached", "MEETING_QUARTER_LIMIT", 409);
+          }
+
+          const meeting = await db.meetings.create({ data: {
+            connectionId: Number(data.connectionId),
+            externalId: String(data.externalId),
+            activeKey,
+            title: String(data.title || "").trim(),
+            startTime,
+            endTime: data.endTime ? new Date(data.endTime) : null,
+            quarter,
+            reviewStatus: REVIEW.DRAFT,
+            retrievalStatus: RETRIEVAL.WAITING_CONTENT,
+            summary: data.summary ?? null,
+            transcript: data.transcript ?? null,
+            coverage: data.coverage || "{}",
+            generation: 0,
+            deletedAt: null,
+            createdBy: userId ?? null,
+          } });
+          await auditMutation("MEETING_METADATA_CREATED", meeting, userId);
+          return meeting;
+        });
+      } catch (error) {
+        if (error.code === "P2002") throw new MeetingError("This meeting is already imported", "MEETING_DUPLICATE", 409);
+        throw error;
+      }
+    },
+
+    async approve(id, userId) {
+      return transaction(async (db) => {
+        const meeting = await findMeeting(db, id);
+        if (meeting.retrievalStatus !== RETRIEVAL.AVAILABLE) {
+          throw new MeetingError("Meeting content must be indexed before approval", "MEETING_NOT_AVAILABLE", 409);
+        }
+        const [sources, jobs] = await Promise.all([
+          db.meeting_sources.findMany({ where: { meetingId: meeting.id, deletedAt: null } }),
+          db.meeting_jobs.findMany({ where: { meetingId: meeting.id } }),
+        ]);
+        if (sources.some((source) => source.parseStatus !== "SUCCESS") ||
+            jobs.some((job) => ["CREATED", "PROCESSING", "FAILED"].includes(job.status) || job.error)) {
+          throw new MeetingError("Meeting has unresolved source or job errors", "MEETING_UNRESOLVED_WORK", 409);
+        }
+        const updated = await db.meetings.update({ where: { id: meeting.id }, data: {
+          reviewStatus: REVIEW.APPROVED, reviewedBy: userId ?? null, reviewedAt: now(),
+        } });
+        await auditMutation("MEETING_APPROVED", updated, userId);
+        return updated;
+      });
+    },
+
+    async softDelete(id, userId) {
+      return transaction(async (db) => {
+        const meeting = await findMeeting(db, id);
+        if (meeting.reviewStatus !== REVIEW.DRAFT) {
+          throw new MeetingError("Only draft meetings can be deleted", "MEETING_IMMUTABLE", 409);
+        }
+        const deletedAt = now();
+        const updated = await db.meetings.update({ where: { id: meeting.id }, data: {
+          activeKey: null, retrievalStatus: RETRIEVAL.DELETED, deletedAt,
+        } });
+        await db.meeting_sources.updateMany({
+          where: { meetingId: meeting.id, deletedAt: null }, data: { deletedAt, activeHash: null },
+        });
+        await db.meeting_chunks.deleteMany({ where: { meetingId: meeting.id } });
+        await auditMutation("MEETING_DELETED", updated, userId);
+        return updated;
+      });
+    },
+
+    async revise(id, changes = {}, userId) {
+      const hasSummary = Object.prototype.hasOwnProperty.call(changes, "summary");
+      const hasTranscript = Object.prototype.hasOwnProperty.call(changes, "transcript");
+      if (!hasSummary && !hasTranscript) throw new MeetingError("A summary or transcript revision is required", "EMPTY_REVISION");
+      return transaction(async (db) => {
+        const meeting = await findMeeting(db, id);
+        if (meeting.reviewStatus !== REVIEW.DRAFT && meeting.reviewStatus !== REVIEW.APPROVED) {
+          throw new MeetingError("Meeting cannot be revised in its current state", "MEETING_IMMUTABLE", 409);
+        }
+        const generation = Number(meeting.generation || 0) + 1;
+        const updated = await db.meetings.update({ where: { id: meeting.id }, data: {
+          ...(hasSummary ? { summary: changes.summary } : {}),
+          ...(hasTranscript ? { transcript: changes.transcript } : {}),
+          generation,
+          reviewStatus: REVIEW.DRAFT,
+          reviewedBy: null,
+          reviewedAt: null,
+          retrievalStatus: RETRIEVAL.INDEXING,
+        } });
+        await db.meeting_jobs.create({ data: {
+          kind: "REINDEX", meetingId: meeting.id, connectionId: meeting.connectionId,
+          externalId: meeting.externalId, activeKey: `REINDEX:${meeting.id}:${generation}`,
+          quarter: meeting.quarter, payload: JSON.stringify({ generation }), status: "CREATED", attempts: 0,
+        } });
+        await auditMutation("MEETING_REVISED", updated, userId, { generation });
+        return updated;
+      });
+    },
+  };
+}
+
+module.exports = { createMeetingService };

+const { LIMITS, REVIEW, RETRIEVAL } = require("./constants");
+const { MeetingError } = require("./errors");
+
+function shanghaiParts(date) {
+  const value = date instanceof Date ? date : new Date(date);
+  if (Number.isNaN(value.getTime())) throw new MeetingError("A valid date is required", "INVALID_DATE");
+  const parts = new Intl.DateTimeFormat("en-US", {
+    timeZone: "Asia/Shanghai", year: "numeric", month: "numeric",
+  }).formatToParts(value);
+  return Object.fromEntries(parts.map(({ type, value: part }) => [type, Number(part)]));
+}
+
+function quarterFor(date) {
+  const { year, month } = shanghaiParts(date);
+  return `${year}-Q${Math.floor((month - 1) / 3) + 1}`;
+}
+
+function validateScope(scope = {}) {
+  const ids = Array.isArray(scope) ? scope : (scope.meetingIds || scope.meetings || []);
+  const uniqueIds = new Set(ids.map((meeting) => typeof meeting === "object" ? meeting.id : meeting));
+  if (uniqueIds.size > LIMITS.selectedMeetings) {
+    throw new MeetingError("Select no more than ten meetings", "MEETING_SCOPE_LIMIT");
+  }
+
+  let duration = scope.durationMs;
+  const from = scope.from || scope.startTime;
+  const to = scope.to || scope.endTime;
+  if (duration == null && from != null && to != null) duration = new Date(to).getTime() - new Date(from).getTime();
+  if (duration != null && (!Number.isFinite(Number(duration)) || Number(duration) < 0)) {
+    throw new MeetingError("A valid transcript time interval is required", "INVALID_TRANSCRIPT_SCOPE");
+  }
+  if (duration > LIMITS.transcriptDurationMs) {
+    throw new MeetingError("Transcript selection cannot exceed three hours", "TRANSCRIPT_SCOPE_LIMIT");
+  }
+  return scope;
+}
+
+function assertQueryable(meetings) {
+  const unavailable = (meetings || []).filter((meeting) => meeting.retrievalStatus !== RETRIEVAL.AVAILABLE);
+  if (unavailable.length) {
+    throw new MeetingError("Every selected meeting must be available before querying", "MEETING_UNAVAILABLE", 409);
+  }
+  return meetings;
+}
+
+function assertSourceDeletion(meeting, source) {
+  if (source.sourceType === "TENCENT_ORIGINAL") {
+    throw new MeetingError("Original meeting sources cannot be deleted", "SOURCE_IMMUTABLE", 409);
+  }
+  if (meeting.reviewStatus !== REVIEW.DRAFT || !["SUPPLEMENT_FILE", "SUPPLEMENT_TEXT"].includes(source.sourceType)) {
+    throw new MeetingError("Only supplemental sources in a draft can be deleted", "SOURCE_IMMUTABLE", 409);
+  }
+  return true;
+}
+
+function assertFileLimits(existing = [], incoming = []) {
+  const current = Array.isArray(existing) ? existing : [];
+  const additions = Array.isArray(incoming) ? incoming : [incoming];
+  const files = [...current.filter((file) => !file.deletedAt), ...additions];
+  if (additions.some((file) => Number(file.bytes || 0) > LIMITS.fileBytes)) {
+    throw new MeetingError("Each file must be 100 MB or smaller", "FILE_SIZE_LIMIT", 413);
+  }
+  if (files.length > LIMITS.filesPerMeeting) {
+    throw new MeetingError("A meeting can have no more than 20 active files", "MEETING_FILE_COUNT_LIMIT", 413);
+  }
+  if (files.reduce((sum, file) => sum + Number(file.bytes || 0), 0) > LIMITS.meetingBytes) {
+    throw new MeetingError("A meeting can contain no more than 500 MB of files", "MEETING_SIZE_LIMIT", 413);
+  }
+  return true;
+}
+
+module.exports = {
+  LIMITS,
+  REVIEW,
+  RETRIEVAL,
+  quarterFor,
+  validateScope,
+  assertQueryable,
+  assertSourceDeletion,
+  assertFileLimits,
+};
